"""
Final Implementation for Thoracic Pathology Detection
Focus: Minimizing False Negatives using Asymmetric Cost Weighting
"""

import os
import torch
import torch.nn as nn
import torch.optim as optim
import pandas as pd
import numpy as np
from torchvision import models, transforms
from PIL import Image
from tqdm import tqdm

# --- 1. GLOBAL SETTINGS ---
IMG_DIR = '/kaggle/input/competitions/26-t-1-dl-gen-ainppe-1/images'
TRAIN_CSV = '/kaggle/input/competitions/26-t-1-dl-gen-ainppe-1/train.csv'
TEST_CSV = '/kaggle/input/competitions/26-t-1-dl-gen-ainppe-1/test.csv'
CLASSES = [
    'Atelectasis', 'Cardiomegaly', 'Consolidation', 'Edema', 'Effusion', 
    'Emphysema', 'Fibrosis', 'Hernia', 'Infiltration', 'Mass', 'Nodule', 
    'Pleural_Thickening', 'Pneumonia', 'Pneumothorax', 'Pneumoperitoneum', 
    'Pneumomediastinum', 'Subcutaneous Emphysema', 'Tortuous Aorta', 
    'Calcification of the Aorta', 'No Finding'
]
NO_FINDING_IDX = CLASSES.index('No Finding')
DEVICE = "cuda" if torch.cuda.is_available() else "cpu"

# --- 2. IMAGE UTILITIES ---
def load_and_preprocess(img_id, is_training=True):
    path = os.path.join(IMG_DIR, img_id)
    img = Image.open(path).convert('RGB')
    
    # Pre-trained models expect specific normalization
    transform_list = [transforms.Resize((224, 224))]
    if is_training:
        transform_list.append(transforms.RandomHorizontalFlip())
    transform_list.extend([
        transforms.ToTensor(),
        transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])
    ])
    
    return transforms.Compose(transform_list)(img)

# --- 3. MODEL INITIALIZATION ---
# Load EfficientNet-B0 with ImageNet weights
model = models.efficientnet_b0(weights='DEFAULT')
# Replace the head for 20-class classification
model.classifier[1] = nn.Linear(model.classifier[1].in_features, len(CLASSES))
model = model.to(DEVICE)

# --- 4. TRAINING LOGIC ---
optimizer = optim.AdamW(model.parameters(), lr=1e-4)
criterion = nn.CrossEntropyLoss(reduction='none') # 'none' allows sample-wise weighting

def train_model(train_df, epochs=2, batch_size=32):
    print(f"Training on {DEVICE}...")
    for epoch in range(epochs):
        model.train()
        # Shuffle for better convergence
        train_df = train_df.sample(frac=1).reset_index(drop=True)
        
        for i in tqdm(range(0, len(train_df), batch_size)):
            batch = train_df.iloc[i : i + batch_size]
            
            # Manual batch assembly (No classes used)
            imgs = torch.stack([load_and_preprocess(r['id']) for _, r in batch.iterrows()]).to(DEVICE)
            labels = torch.tensor([np.argmax(r[CLASSES].values) for _, r in batch.iterrows()]).to(DEVICE)
            
            optimizer.zero_grad()
            outputs = model(imgs)
            
            # ASYMMETRIC COST CALCULATION
            # Penalize missing a disease (True label != 'No Finding') by factor of 5
            loss_raw = criterion(outputs, labels)
            weights = torch.ones_like(labels, dtype=torch.float)
            weights[labels != NO_FINDING_IDX] = 5.0
            
            loss = (loss_raw * weights).mean()
            loss.backward()
            optimizer.step()

# --- 5. INFERENCE & SUBMISSION ---
def generate_submission(test_path):
    test_df = pd.read_csv(test_path)
    model.eval()
    results = []
    
    with torch.no_grad():
        for i in tqdm(range(0, len(test_df), 32)):
            batch = test_df.iloc[i : i + 32]
            imgs = torch.stack([load_and_preprocess(r['id'], False) for _, r in batch.iterrows()]).to(DEVICE)
            
            probs = torch.softmax(model(imgs), dim=1)
            
            # REDUCE NO-FINDING PROBABILITY
            # This forces the model to choose a disease in borderline cases
            # helps avoid the heavy -5 penalty
            probs[:, NO_FINDING_IDX] *= 0.85
            preds = torch.argmax(probs, dim=1).cpu().numpy()
            
            for idx, p_idx in enumerate(preds):
                row = {cls: 0 for cls in CLASSES}
                row['id'] = batch.iloc[idx]['id']
                row[CLASSES[p_idx]] = 1
                results.append(row)
                
    pd.DataFrame(results)[['id'] + CLASSES].to_csv('submission.csv', index=False)

# --- EXECUTION ---
if __name__ == "__main__":
    train_data = pd.read_csv(TRAIN_CSV)
    train_model(train_data)
    generate_submission(TEST_CSV)
